# ── Static files in production ─────────────────────────────
# ⚠️  Django refuses to serve static files when DEBUG is off, and this deployment
#     has no separate web server in front of them: Passenger owns the whole
#     domain, so the React bundle and the admin's CSS both arrive as ordinary
#     Django requests. Without WhiteNoise every one of them is a 404 — the site
#     loads as unstyled HTML and the frontend never boots.
whitenoise==6.7.0

# ── Media ──────────────────────────────────────────────────
Pillow==11.2.1

# ── Spreadsheets — bulk catalogue import ───────────────────
# ⚠️  It reads and writes `.xlsx`, and it is used in `read_only=True` mode alone.
#
#     The ordinary mode builds every cell of the workbook as an object before
#     the first row is read: a 10,000-row sheet with forty columns is hundreds
#     of megabytes of Python objects on a shared host with a hard memory cap.
#     `read_only=True` streams the rows instead — see imports/reader.py.
#
# ⚠️  And **no pandas.** It would arrive with numpy for work that is a loop over
#     rows, and its `float64` inference silently turns a price into a binary
#     float — which core/money.py forbids in nine words.
openpyxl==3.1.5

# ── Translation ────────────────────────────────────────────
# ⚠️  A build tool that production nonetheless needs — so it stays here, not in `dev.txt`.
#
#     `.mo` files are gitignored (`.gitignore:118`); only `locale/ar/LC_MESSAGES/django.po`
#     is committed. So the compiled catalogue does not arrive with the upload — it is
#     produced on the server. Django's own `compilemessages` calls GNU gettext's
#     `msgfmt`, which shared hosting rarely provides, and this is the substitute:
#
#         pybabel compile -d locale -D django
#
#     Without it every translated string silently falls back to its msgid.
babel==2.18.0
